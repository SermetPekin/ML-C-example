# CIFAR-10 Image Classification Configuration - Adam Optimizer
# This configuration shows how to use the Adam optimizer instead of SGD

[dataset]
# Data file paths (generate by flattening 32x32x3 images to 3072D vectors)
train_images = examples/cifar10/data/cifar10_train_images.mat
train_labels = examples/cifar10/data/cifar10_train_labels.mat
test_images = examples/cifar10/data/cifar10_test_images.mat
test_labels = examples/cifar10/data/cifar10_test_labels.mat

# Dataset dimensions
train_size = 50000        # 50000 training samples
test_size = 10000         # 10000 test samples
input_size = 3072         # 32x32x3 = 3072 flattened features
output_size = 10          # 10 classes
label_format = integer_indices  # Labels are 0-9

[training]
# Training hyperparameters - more epochs for harder task
epochs = 50
batch_size = 128
# Adam typically uses smaller learning rates than SGD
learning_rate = 0.0005

# Use Adam optimizer (options: sgd, adam)
optimizer = adam

# Adam hyperparameters (only used when optimizer = adam)
adam_beta1 = 0.9        # Decay rate for 1st moment (usually 0.9)
adam_beta2 = 0.999      # Decay rate for 2nd moment (usually 0.999)
adam_epsilon = 1e-8     # Numerical stability constant

[architecture]
# Deeper network for image classification
# Input (3072) -> Hidden1 (512) -> Hidden2 (256) -> Output (10)

# First hidden layer: 3072 features -> 512 units with ReLU
layer = dense 3072 512 relu

# Second hidden layer: 512 units -> 256 units with ReLU
layer = dense 512 256 relu

# Output layer: 256 units -> 10 classes with softmax
layer = dense 256 10 softmax
